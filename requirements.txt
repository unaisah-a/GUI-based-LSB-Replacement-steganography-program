# INF2005 ACW1 dependencies.
#
# Every entry is pinned so the demo machine resolves identical builds. Install
# into a virtual environment rather than the system interpreter:
#
#   python -m venv .venv
#   .venv\Scripts\python -m pip install -r requirements.txt    (Windows)
#   .venv/bin/python -m pip install -r requirements.txt        (macOS/Linux)
#
# Keep this file merge-friendly: add entries under your own section rather than
# editing someone else's, and leave the section headers in place.

# --- Shared numerics --------------------------------------------------------
# Used by the image layer (vectorised bit and metric arithmetic; Requirement
# 13.8 sets a 36-million-element budget that rules out Python loops), by the
# audio layer for sample handling, and by the video layer for frame handling.
numpy==2.1.3

# --- Image steganography and image steganalysis (Member 2) ------------------
# Pillow is used for PNG only. BMP is encoded and decoded directly in
# app/stego/image_io.py, because Pillow's BMP writer discards the alpha channel
# and stores grayscale as an 8-bit palette.
Pillow==11.0.0

# --- Audio steganography and robust embedding (Member 3) --------------------
# soundfile: read and write WAV/PCM audio.
# scipy: scientific computing helpers for audio analysis.
soundfile==0.12.1
scipy==1.14.1

# --- Cryptography and payload security (Member 1) ---------------------------
# cryptography: RSA-3072 key handling, PSS signatures, SHA-256 hashing,
# AES-256-GCM authenticated encryption and the scrypt KDF. Imported by
# app/crypto/signatures.py, key_manager.py, hashing.py, encryption.py and
# envelope.py.
cryptography==43.0.1

# --- Verification, attacks and video (Member 5) -----------------------------
# opencv-python: lossless video frame read/write for app/stego/video_stego.py
# and the frame-level media comparison in app/verification/media_compare.py.
# The FFV1 codec ships inside the OpenCV wheel, so no external ffmpeg binary is
# required on the demo machine.
opencv-python==4.10.0.84

# --- GUI and system integration (Member 4) ----------------------------------
# PySide6: Qt bindings for the desktop GUI, including QtMultimedia for audio
# and video playback.
# Note: the analysis layer must not import PySide6 (Requirement 12.2).
PySide6==6.8.0

# --- Testing (shared) -------------------------------------------------------
# hypothesis drives the property-based tests required by Requirement 15.
# pytest-qt drives the GUI widget smoke tests.
pytest==8.3.3
hypothesis==6.115.5
pytest-qt==4.4.0
